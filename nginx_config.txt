# Nginx 配置示例
# 请在服务器上修改 /etc/nginx/sites-available/default 文件

server {
    server_name wulara.top;

    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        # 添加这一行，确保 Django 能识别 HTTPS 请求
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 静态文件配置
    location /media/ {
        alias /path/to/your/project/media/;
        expires 30d;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/wulara.top/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/wulara.top/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = wulara.top) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    server_name wulara.top;
    return 404; # managed by Certbot
}
